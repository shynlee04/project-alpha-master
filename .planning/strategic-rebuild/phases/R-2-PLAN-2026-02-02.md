### R-2-01: Create Storage Compatibility Layer

**Description:** Create adapter that reads old data format but writes new format

**File:** `src/platform/core/storage-gateway.ts`

```typescript
/**
 * StorageGateway - Compatibility layer for database access
 * 
 * READS: Both old (with workspaceId) and new (projectId only) records
 * WRITES: Only new format (projectId only, no workspaceId)
 * 
 * This enables gradual migration without data loss.
 */
import { db } from '@/infrastructure/persistence/dexie-db';
import type { Project } from '../types';

export class StorageGateway {
  /**
   * Load project - strips deprecated fields on read
   */
  async loadProject(projectId: string): Promise<Project | null> {
    const rawProject = await db.projects.get(projectId);
    if (!rawProject) return null;
    
    // Strip deprecated fields
    const { workspaceId, workspaceBindings, ...cleanProject } = rawProject as any;
    
    return cleanProject as Project;
  }

  /**
   * Save project - only saves clean fields
   */
  async saveProject(project: Project): Promise<void> {
    // Ensure no deprecated fields
    const cleanProject = {
      id: project.id,
      name: project.name,
      description: project.description,
      storageType: project.storageType,
      directoryHandle: project.directoryHandle,
      settings: project.settings,
      createdAt: project.createdAt,
      updatedAt: new Date(),
    };
    
    await db.projects.put(cleanProject);
  }

  /**
   * Load all projects - strips deprecated fields
   */
  async loadAllProjects(): Promise<Project[]> {
    const rawProjects = await db.projects.toArray();
    return rawProjects.map((raw: any) => {
      const { workspaceId, workspaceBindings, ...clean } = raw;
      return clean as Project;
    });
  }
}

export const storageGateway = new StorageGateway();
```

**Validation:** 
- Old projects still load correctly
- New projects save without deprecated fields

### R-2-02: Port Project Store (Clean Types)

**Description:** Create or update project store with clean types

**File:** `src/infrastructure/persistence/stores/project/project-store.ts` (or new location)

**Clean Interface:**
```typescript
/**
 * Clean Project Store - NO workspaceId, NO workspaceBindings
 */
import { create } from 'zustand';
import { devtools, persist } from 'zustand/middleware';
import { storageGateway } from '@/platform/core/storage-gateway';
import type { Project } from '@/platform/types';

interface ProjectState {
  // State
  projects: Map<string, Project>;
  currentProjectId: string | null;
  isLoading: boolean;
  error: Error | null;
  
  // Actions
  loadProjects: () => Promise<void>;
  loadProject: (projectId: string) => Promise<Project | null>;
  createProject: (name: string, storageType: 'fsa' | 'indexeddb') => Promise<Project>;
  updateProject: (projectId: string, updates: Partial<Project>) => Promise<void>;
  deleteProject: (projectId: string) => Promise<void>;
  setCurrentProject: (projectId: string | null) => void;
}

export const useProjectStore = create<ProjectState>()(
  devtools(
    persist(
      (set, get) => ({
        projects: new Map(),
        currentProjectId: null,
        isLoading: false,
        error: null,
        
        loadProjects: async () => {
          set({ isLoading: true, error: null });
          try {
            const projects = await storageGateway.loadAllProjects();
            const projectMap = new Map(projects.map(p => [p.id, p]));
            set({ projects: projectMap, isLoading: false });
          } catch (err) {
            set({ error: err as Error, isLoading: false });
          }
        },
        
        loadProject: async (projectId: string) => {
          const project = await storageGateway.loadProject(projectId);
          if (project) {
            set(state => ({
              projects: new Map(state.projects).set(projectId, project)
            }));
          }
          return project;
        },
        
        createProject: async (name: string, storageType: 'fsa' | 'indexeddb') => {
          const project: Project = {
            id: crypto.randomUUID(),
            name,
            storageType,
            settings: {
              enabledModules: ['monaco', 'notes'],
              defaultModule: 'monaco',
            },
            createdAt: new Date(),
            updatedAt: new Date(),
          };
          
          await storageGateway.saveProject(project);
          
          set(state => ({
            projects: new Map(state.projects).set(project.id, project)
          }));
          
          return project;
        },
        
        updateProject: async (projectId: string, updates: Partial<Project>) => {
          const existing = get().projects.get(projectId);
          if (!existing) throw new Error('Project not found');
          
          const updated: Project = {
            ...existing,
            ...updates,
            updatedAt: new Date(),
          };
          
          await storageGateway.saveProject(updated);
          
          set(state => ({
            projects: new Map(state.projects).set(projectId, updated)
          }));
        },
        
        deleteProject: async (projectId: string) => {
          await storageGateway.deleteProject(projectId);
          
          set(state => {
            const newProjects = new Map(state.projects);
            newProjects.delete(projectId);
            return { projects: newProjects };
          });
        },
        
        setCurrentProject: (projectId: string | null) => {
          set({ currentProjectId: projectId });
        },
      }),
      {
        name: 'project-store',
        partialize: (state) => ({
          currentProjectId: state.currentProjectId,
        }),
      }
    ),
    { name: 'ProjectStore' }
  )
);
```

**Validation:** 
- Store compiles with 0 errors
- Projects load and save correctly
- No workspaceId in store interface
