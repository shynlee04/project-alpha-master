### R-3-01: Create Module Loader System

**Description:** Create unified module loading and registration system

**File:** `src/modules/types.ts`

```typescript
/**
 * Feature Module Interface
 * 
 * All feature modules implement this interface for consistent loading.
 */
export type ModuleType = 'monaco' | 'notes' | 'terminal' | 'preview';

export interface IFeatureModule {
  id: ModuleType;
  name: string;
  icon: string;
  description: string;
  
  // Component to render
  component: React.ComponentType<ModuleProps>;
  
  // Lifecycle
  onMount?: (projectId: string) => void | Promise<void>;
  onUnmount?: () => void;
  onProjectChange?: (newProjectId: string) => void | Promise<void>;
  
  // Feature flags
  requiresProject: boolean;
  supportsOffline: boolean;
}

export interface ModuleProps {
  projectId: string;
  className?: string;
}
```

**File:** `src/modules/loader.ts`

```typescript
/**
 * Module Loader - Lazy loads and manages feature modules
 */
import { lazy, Suspense } from 'react';
import type { IFeatureModule, ModuleType } from './types';

// Lazy load modules for code splitting
const moduleImports = {
  monaco: () => import('./monaco'),
  notes: () => import('./notes'),
  terminal: () => import('./terminal'),
  preview: () => import('./preview'),
};

class ModuleLoader {
  private loadedModules = new Map<ModuleType, IFeatureModule>();
  
  async loadModule(type: ModuleType): Promise<IFeatureModule> {
    if (this.loadedModules.has(type)) {
      return this.loadedModules.get(type)!;
    }
    
    const module = await moduleImports[type]();
    this.loadedModules.set(type, module.default);
    return module.default;
  }
  
  isLoaded(type: ModuleType): boolean {
    return this.loadedModules.has(type);
  }
  
  getLoaded(): IFeatureModule[] {
    return Array.from(this.loadedModules.values());
  }
}

export const moduleLoader = new ModuleLoader();
```

**File:** Create `src/modules/ModulePanel.tsx`

```typescript
import { Suspense, lazy } from 'react';
import type { ModuleType } from './types';

const modules = {
  monaco: lazy(() => import('./monaco').then(m => ({ default: m.default.component }))),
  notes: lazy(() => import('./notes').then(m => ({ default: m.default.component }))),
  terminal: lazy(() => import('./terminal').then(m => ({ default: m.default.component }))),
  preview: lazy(() => import('./preview').then(m => ({ default: m.default.component }))),
};

interface ModulePanelProps {
  projectId: string;
  activeModule: ModuleType;
}

export function ModulePanel({ projectId, activeModule }: ModulePanelProps) {
  const ModuleComponent = modules[activeModule];
  
  return (
    <Suspense fallback={<div>Loading module...</div>}>
      <ModuleComponent projectId={projectId} />
    </Suspense>
  );
}
```
