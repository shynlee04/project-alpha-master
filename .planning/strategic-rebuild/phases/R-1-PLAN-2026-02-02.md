### R-1-03: Create PlatformLayout Component

**Description:** Update PlatformLayout to use actual operator components

**File:** `src/platform/core/platform-layout.tsx`

```typescript
/**
 * PlatformLayout - The main layout for project routes
 * 
 * KEY DESIGN:
 * - FileTree and Chat are HARDCODED here (not store-driven)
 * - Only the middle panel uses activity-bar store
 * - No hydration waiting - if project loads, layout renders
 */
import React, { ReactNode } from 'react';
import { usePlatform } from './platform-context';
import { FileTreeOperator } from '../operators/filetree';
import { ChatOperator } from '../operators/chat';
import './platform-layout.css';

interface PlatformLayoutProps {
  children: ReactNode;
}

export function PlatformLayout({ children }: PlatformLayoutProps) {
  const { isLoading, error, projectId } = usePlatform();

  if (isLoading) {
    return (
      <div className="platform-layout platform-layout--loading">
        <div className="platform-loading">Loading project...</div>
      </div>
    );
  }

  if (error) {
    return (
      <div className="platform-layout platform-layout--error">
        <div className="platform-error">
          <h2>Error Loading Project</h2>
          <p>{error.message}</p>
        </div>
      </div>
    );
  }

  if (!projectId) {
    return (
      <div className="platform-layout platform-layout--error">
        <div className="platform-error">
          <h2>No Project Selected</h2>
          <p>Please select a project from the hub.</p>
        </div>
      </div>
    );
  }

  return (
    <div className="platform-layout">
      {/* LEFT: FileTree Operator - ALWAYS HERE */}
      <aside className="platform-operator platform-operator--filetree">
        <FileTreeOperator projectId={projectId} />
      </aside>

      {/* CENTER: Module Panel - store-driven */}
      <main className="platform-modules">
        {children}
      </main>

      {/* RIGHT: Chat Operator - ALWAYS HERE */}
      <aside className="platform-operator platform-operator--chat">
        <ChatOperator projectId={projectId} />
      </aside>
    </div>
  );
}
```

**Validation:** Layout renders both operators

### R-1-04: Wire Operators to Project Route

**Description:** Ensure project route uses PlatformLayout with operators

**File:** `src/routes/$projectId/route.tsx` (update from R-0)

```typescript
import { createFileRoute, Outlet } from '@tanstack/react-router';
import { PlatformProvider } from '@/platform/core/platform-context';
import { PlatformLayout } from '@/platform/core/platform-layout';

export const Route = createFileRoute('/$projectId')({
  component: ProjectRoute,
});

function ProjectRoute() {
  const { projectId } = Route.useParams();
  
  return (
    <PlatformProvider projectId={projectId}>
      <PlatformLayout>
        <Outlet />
      </PlatformLayout>
    </PlatformProvider>
  );
}
```

### R-1-05: Remove Operators from Activity-Bar Store

**Description:** Simplify activity-bar store to only manage Feature Modules

**Current Problem:**
- Activity-bar store manages both operators AND modules
- Operators depend on store hydration → hydration race bugs

**Solution:**
- Remove FileTree and Chat from activity-bar store
- Store only manages: Monaco, Notes, Terminal, Preview
- Operators are hardcoded in PlatformLayout

**File:** Update `src/infrastructure/persistence/stores/activity-bar-store.ts` (or equivalent)

**Changes:**
```typescript
// REMOVE these from store:
// - 'filetree' panel
// - 'chat' panel

// KEEP only Feature Modules:
type ModulePanel = 'monaco' | 'notes' | 'terminal' | 'preview';

interface ActivityBarState {
  activeModules: ModulePanel[];
  defaultModule: ModulePanel;
  // ... rest
}
```

**Validation:** 
- Store compiles
- Operators still work (not from store)
- Activity bar only shows module toggles
