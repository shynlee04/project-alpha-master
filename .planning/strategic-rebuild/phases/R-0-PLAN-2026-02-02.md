```typescript
/**
 * Platform capabilities detected at runtime
 * NO workspaceId - use projectId only
 */
export interface PlatformCapabilities {
  platform: 'desktop' | 'tablet' | 'mobile';
  hasFileSystemAccess: boolean;
  hasWebContainer: boolean;
}

export type StorageType = 'fsa' | 'indexeddb';

export function detectPlatform(): PlatformCapabilities {
  const hasFileSystemAccess = typeof window !== 'undefined' && 
    'showDirectoryPicker' in window;
  
  const isMobile = typeof window !== 'undefined' && 
    /Android|webOS|iPhone|iPad|iPod/i.test(navigator.userAgent);
  
  const isTablet = typeof window !== 'undefined' &&
    /iPad|Android/i.test(navigator.userAgent) && 
    window.innerWidth >= 768;

  return {
    platform: isMobile ? 'mobile' : isTablet ? 'tablet' : 'desktop',
    hasFileSystemAccess,
    hasWebContainer: hasFileSystemAccess, // WebContainer requires FSA
  };
}
```

### R-0-03: Create PlatformProvider Context

**Description:** Create the context that provides platform state to all components

**File:** `src/platform/core/platform-context.tsx`

```typescript
/**
 * PlatformProvider - Core context for platform layer
 * 
 * This is SIMPLE - no complex hydration, no store-driven state
 * Platform is either loading or ready, that's it.
 */
import React, { createContext, useContext, useState, useEffect, ReactNode } from 'react';
import type { Project, PlatformCapabilities } from '../types';
import { detectPlatform } from '../types';

interface PlatformContextValue {
  // Platform
  platform: PlatformCapabilities;
  
  // Project
  project: Project | null;
  projectId: string | null;
  isLoading: boolean;
  error: Error | null;
  
  // Actions
  loadProject: (projectId: string) => Promise<void>;
}

const PlatformContext = createContext<PlatformContextValue | null>(null);

interface PlatformProviderProps {
  projectId: string;
  children: ReactNode;
}

export function PlatformProvider({ projectId, children }: PlatformProviderProps) {
  const [platform] = useState(() => detectPlatform());
  const [project, setProject] = useState<Project | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  const loadProject = async (id: string) => {
    setIsLoading(true);
    setError(null);
    try {
      // TODO: R-2 will implement actual loading from Dexie
      // For now, just simulate
      await new Promise(resolve => setTimeout(resolve, 100));
      // setProject(loadedProject);
    } catch (err) {
      setError(err instanceof Error ? err : new Error('Failed to load project'));
    } finally {
      setIsLoading(false);
    }
  };

  useEffect(() => {
    if (projectId) {
      loadProject(projectId);
    }
  }, [projectId]);

  const value: PlatformContextValue = {
    platform,
    project,
    projectId,
    isLoading,
    error,
    loadProject,
  };

  return (
    <PlatformContext.Provider value={value}>
      {children}
    </PlatformContext.Provider>
  );
}

export function usePlatform() {
  const context = useContext(PlatformContext);
  if (!context) {
    throw new Error('usePlatform must be used within PlatformProvider');
  }
  return context;
}

export function usePlatformSafe() {
  return useContext(PlatformContext);
}
```

```typescript
/**
 * Clean Project Route
 * 
 * This route uses the new PlatformProvider and has NO hydration dependencies.
 * Platform Operators (FileTree, Chat) are rendered directly, not from store.
 */
import { createFileRoute, Outlet } from '@tanstack/react-router';
import { PlatformProvider } from '@/platform/core/platform-context';
import { PlatformLayout } from '@/platform/core/platform-layout';

export const Route = createFileRoute('/$projectId')({
  component: ProjectRoute,
});

function ProjectRoute() {
  const { projectId } = Route.useParams();
  
  return (
    <PlatformProvider projectId={projectId}>
      <PlatformLayout>
        <Outlet />
      </PlatformLayout>
    </PlatformProvider>
  );
}
```

### R-0-05: Create Simple Layout (No Hydration)

```typescript
/**
 * PlatformLayout - The main layout for project routes
 * 
 * KEY DESIGN:
 * - FileTree and Chat are HARDCODED here (not store-driven)
 * - Only the middle panel uses activity-bar store
 * - No hydration waiting - if project loads, layout renders
 */
import React, { ReactNode } from 'react';
import { usePlatform } from './platform-context';

interface PlatformLayoutProps {
  children: ReactNode;
}

export function PlatformLayout({ children }: PlatformLayoutProps) {
  const { isLoading, error, projectId } = usePlatform();

  if (isLoading) {
    return (
      <div className="platform-layout platform-layout--loading">
        <div className="platform-loading">Loading project...</div>
      </div>
    );
  }

  if (error) {
    return (
      <div className="platform-layout platform-layout--error">
        <div className="platform-error">
          <h2>Error Loading Project</h2>
          <p>{error.message}</p>
        </div>
      </div>
    );
  }

  return (
    <div className="platform-layout">
      {/* LEFT: FileTree Operator - ALWAYS HERE */}
      <aside className="platform-operator platform-operator--filetree">
        {/* TODO: R-1 will add actual FileTree */}
        <div className="operator-placeholder">
          FileTree Operator (R-1)
          <br />
          Project: {projectId}
        </div>
      </aside>

      {/* CENTER: Module Panel - store-driven */}
      <main className="platform-modules">
        {children}
      </main>

      {/* RIGHT: Chat Operator - ALWAYS HERE */}
      <aside className="platform-operator platform-operator--chat">
        {/* TODO: R-1 will add actual Chat */}
        <div className="operator-placeholder">
          Chat Operator (R-1)
        </div>
      </aside>
    </div>
  );
}
```
