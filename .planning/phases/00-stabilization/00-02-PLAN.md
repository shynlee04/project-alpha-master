       ```typescript
       // @/infrastructure/persistence/stores/project/module-settings-slice.ts
       import type { StateCreator } from 'zustand';
       import type { ModuleType, ProjectSettings } from '@/domain/entities/project';
       
       export interface ModuleSettingsSlice {
         // State
         enabledModules: ModuleType[];
         defaultModule: ModuleType;
         
         // Actions
         setEnabledModules: (modules: ModuleType[]) => void;
         toggleModule: (module: ModuleType) => void;
         setDefaultModule: (module: ModuleType) => void;
       }
       
       export const createModuleSettingsSlice: StateCreator<
         ModuleSettingsSlice,
         [],
         [],
         ModuleSettingsSlice
       > = (set) => ({
         enabledModules: ['ide'],
         defaultModule: 'ide',
         
         setEnabledModules: (modules) => set({ enabledModules: modules }),
         toggleModule: (module) => set((state) => ({
           enabledModules: state.enabledModules.includes(module)
             ? state.enabledModules.filter(m => m !== module)
             : [...state.enabledModules, module]
         })),
         setDefaultModule: (module) => set({ defaultModule: module }),
       });
       ```
