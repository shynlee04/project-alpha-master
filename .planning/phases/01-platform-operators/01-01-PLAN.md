// src/domain/types/domain-events.ts
export type DomainEventType =
  | 'file:created' | 'file:updated' | 'file:deleted' | 'file:synced'
  | 'project:created' | 'project:deleted' | 'project:switched'
  | 'thread:created' | 'thread:updated' | 'thread:deleted'
  | 'tool:executed' | 'tool:approved' | 'tool:rejected';

export interface DomainEvent<T = unknown> {
  type: DomainEventType;
  payload: T;
  timestamp: number;
  source: string; // operator/service that emitted

// Typed payloads for each event

Then implement the event bus:

```typescript
// src/infrastructure/events/domain-event-bus.ts
type EventHandler<T> = (event: DomainEvent<T>) => void;

class DomainEventBus {
  private handlers = new Map<DomainEventType, Set<EventHandler<unknown>>>();
  
  emit<T>(type: DomainEventType, payload: T, source: string): void {
    const event: DomainEvent<T> = { type, payload, timestamp: Date.now(), source };
    const handlers = this.handlers.get(type);
    handlers?.forEach(h => h(event as DomainEvent<unknown>));
  }
  
  on<T>(type: DomainEventType, handler: EventHandler<T>): () => void {
    if (!this.handlers.has(type)) this.handlers.set(type, new Set());
    this.handlers.get(type)!.add(handler as EventHandler<unknown>);
    return () => this.handlers.get(type)?.delete(handler as EventHandler<unknown>);
  }
  
  off(type: DomainEventType, handler: EventHandler<unknown>): void {
    this.handlers.get(type)?.delete(handler);
  }
}

export const domainEventBus = new DomainEventBus();
export { DomainEventBus };
```

Key points:
- Simple pub/sub pattern, no external dependencies
- Returns unsubscribe function from `on()` for cleanup
- Typed events prevent typos and enable autocomplete
- Source tracking helps debug event chains
