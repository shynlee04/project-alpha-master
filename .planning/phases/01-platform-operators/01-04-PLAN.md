// src/domain/services/thread-service.ts
import { nanoid } from 'nanoid';
import { db } from '@/infrastructure/persistence/dexie-db';
import { domainEventBus } from '@/infrastructure/events/domain-event-bus';
import type { IThreadService } from '@/domain/interfaces/thread-service.interface';
import type { Thread, ThreadMessage } from '@/domain/schemas/thread.schema';

class ThreadService implements IThreadService {
  async createThread(projectId: string, title = 'New Chat'): Promise<Thread> {
    const thread: Thread = {
      id: nanoid(),
      projectId,
      title,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      messages: [],
    };
    
    await db.threads.put(thread);
    domainEventBus.emit('thread:created', { threadId: thread.id, projectId }, 'ThreadService');
    
    return thread;
  }
  
  async getThread(threadId: string): Promise<Thread | null> {
    return db.threads.get(threadId) ?? null;
  }
  
  async listThreads(projectId: string): Promise<Thread[]> {
    return db.threads.where('projectId').equals(projectId).toArray();
  }
  
  async deleteThread(threadId: string): Promise<void> {
    await db.threads.delete(threadId);
    domainEventBus.emit('thread:deleted', { threadId }, 'ThreadService');
  }
  
  async addMessage(threadId: string, message: Omit<ThreadMessage, 'id' | 'createdAt'>): Promise<ThreadMessage> {
    const thread = await this.getThread(threadId);
    if (!thread) throw new Error(`Thread not found: ${threadId}`);
    
    const fullMessage: ThreadMessage = {
      ...message,
      id: nanoid(),
      createdAt: new Date().toISOString(),
    };
    
    thread.messages.push(fullMessage);
    thread.updatedAt = new Date().toISOString();
    await db.threads.put(thread);
    
    domainEventBus.emit('thread:updated', { threadId }, 'ThreadService');
    
    return fullMessage;
  }
  
  async getMessages(threadId: string): Promise<ThreadMessage[]> {
    const thread = await this.getThread(threadId);
    return thread?.messages ?? [];
  }
  
  async updateMessage(threadId: string, messageId: string, updates: Partial<ThreadMessage>): Promise<void> {
    const thread = await this.getThread(threadId);
    if (!thread) throw new Error(`Thread not found: ${threadId}`);
    
    const message = thread.messages.find(m => m.id === messageId);
    if (message) {
      Object.assign(message, updates);
      thread.updatedAt = new Date().toISOString();
      await db.threads.put(thread);
    }
  }
}

export const threadService = new ThreadService();
export { ThreadService };
```

Key points:
- Uses Dexie for persistence (L3 layer)
- Emits domain events for cross-operator communication
- projectId scope (NO workspaceId)
- nanoid for ID generation

import type { IPlatformOperator } from '@/domain/interfaces/operator.interface';
import { threadService } from '@/domain/services/thread-service';
import { domainEventBus } from '@/infrastructure/events/domain-event-bus';

class ChatOperator implements IPlatformOperator {
  readonly name = 'Chat-Cascade';
  readonly isOperator = true as const;
  
  private unsubscribers: (() => void)[] = [];
  private activeThreadId: string | null = null;
  
  async init(): Promise<void> {
    // Subscribe to project switch to load appropriate threads
    this.unsubscribers.push(
      domainEventBus.on('project:switched', (e) => this.handleProjectSwitch(e.payload))
    );
    
    // Subscribe to file events for potential RAG context (future)
    this.unsubscribers.push(
      domainEventBus.on('file:updated', (e) => this.handleFileUpdate(e.payload))
    );
  }
  
  async destroy(): Promise<void> {
    this.unsubscribers.forEach(unsub => unsub());
    this.unsubscribers = [];
    this.activeThreadId = null;
  }
  
  async healthCheck(): Promise<{ healthy: boolean; message?: string }> {
    try {
      // Verify Dexie connection
      await threadService.listThreads('__health__');
      return { healthy: true };
    } catch (e) {
      return { healthy: false, message: String(e) };
    }
  }
  
  setActiveThread(threadId: string | null): void {
    this.activeThreadId = threadId;
  }
  
  getActiveThread(): string | null {
    return this.activeThreadId;
  }
  
  private handleProjectSwitch(payload: { projectId: string }): void {
    // Clear active thread on project switch
    this.activeThreadId = null;
  }
  
  private handleFileUpdate(payload: { projectId: string; path: string }): void {
    // Future: Invalidate RAG context cache for this file
  }
}

export const chatOperator = new ChatOperator();
export { ChatOperator };

// src/plugins/chat/hooks/useChat.ts
import { useState, useCallback, useEffect } from 'react';
import { threadService } from '@/domain/services/thread-service';
import { chatOperator } from '../ChatOperator';
import type { Thread, ThreadMessage } from '@/domain/schemas/thread.schema';

export function useProjectChat(projectId: string) {
  const [threads, setThreads] = useState<Thread[]>([]);
  const [activeThread, setActiveThread] = useState<Thread | null>(null);
  const [isLoading, setIsLoading] = useState(false);
  
  // Load threads on mount
  useEffect(() => {
    const load = async () => {
      const list = await threadService.listThreads(projectId);
      setThreads(list);
    };
    load();
  }, [projectId]);
  
  const createThread = useCallback(async (title?: string) => {
    const thread = await threadService.createThread(projectId, title);
    setThreads(prev => [...prev, thread]);
    setActiveThread(thread);
    chatOperator.setActiveThread(thread.id);
    return thread;
  }, [projectId]);
  
  const selectThread = useCallback(async (threadId: string) => {
    const thread = await threadService.getThread(threadId);
    setActiveThread(thread);
    chatOperator.setActiveThread(threadId);
  }, []);
  
  const sendMessage = useCallback(async (content: string) => {
    if (!activeThread) return;
    
    setIsLoading(true);
    
    // Add user message
    const userMessage = await threadService.addMessage(activeThread.id, {
      role: 'user',
      parts: [{ type: 'text', content }],
    });
    
    setActiveThread(prev => prev ? {
      ...prev,
      messages: [...prev.messages, userMessage],
    } : null);
    
    // TODO: Call AI API and stream response (Phase 2: AI Integration)
    // For now, add placeholder response
    const assistantMessage = await threadService.addMessage(activeThread.id, {
      role: 'assistant',
      parts: [{ type: 'text', content: 'AI response will be implemented in Phase 2.' }],
    });
    
    setActiveThread(prev => prev ? {
      ...prev,
      messages: [...prev.messages, assistantMessage],
    } : null);
    
    setIsLoading(false);
  }, [activeThread]);
  
  return {
    threads,
    activeThread,
    isLoading,
    createThread,
    selectThread,
    sendMessage,
  };
}

// src/plugins/chat/components/ChatPanel.tsx
import { useState } from 'react';
import { useProjectChat } from '../hooks/useChat';
import { useShallow } from 'zustand/react/shallow';
import type { ThreadMessage, MessagePart } from '@/domain/schemas/thread.schema';

interface ChatPanelProps {
  projectId: string;
}

export function ChatPanel({ projectId }: ChatPanelProps) {
  const { threads, activeThread, isLoading, createThread, selectThread, sendMessage } = useProjectChat(projectId);
  const [input, setInput] = useState('');
  
  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!input.trim()) return;
    await sendMessage(input);
    setInput('');
  };
  
  return (
    <div className="flex h-full">
      {/* Thread list sidebar */}
      <div className="w-48 border-r border-border-primary bg-bg-secondary">
        <button
          onClick={() => createThread()}
          className="w-full p-2 text-left hover:bg-bg-tertiary border-b border-border-primary"
          style={{ borderRadius: 0 }} // 8-bit: sharp corners
        >
          + New Thread
        </button>
        {threads.map(t => (
          <button
            key={t.id}
            onClick={() => selectThread(t.id)}
            className={`w-full p-2 text-left hover:bg-bg-tertiary ${activeThread?.id === t.id ? 'bg-bg-tertiary' : ''}`}
            style={{ borderRadius: 0 }}
          >
            {t.title}
          </button>
        ))}
      </div>
      
      {/* Chat area */}
      <div className="flex-1 flex flex-col">
        {/* Messages */}
        <div className="flex-1 overflow-y-auto p-4 space-y-4">
          {activeThread?.messages.map(msg => (
            <MessageBubble key={msg.id} message={msg} />
          ))}
          {isLoading && <div className="text-text-muted">Thinking...</div>}
        </div>
        
        {/* Input */}
        <form onSubmit={handleSubmit} className="p-4 border-t border-border-primary">
          <div className="flex gap-2">
            <input
              value={input}
              onChange={e => setInput(e.target.value)}
              placeholder="Type a message..."
              className="flex-1 p-2 bg-bg-secondary border border-border-primary"
              style={{ borderRadius: 0 }}
              disabled={!activeThread || isLoading}
            />
            <button
              type="submit"
              disabled={!activeThread || isLoading}
              className="px-4 py-2 bg-accent-primary text-white disabled:opacity-50"
              style={{ borderRadius: 0, boxShadow: '2px 2px 0 0 var(--shadow-color)' }}
            >
              Send
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

function MessageBubble({ message }: { message: ThreadMessage }) {
  const isUser = message.role === 'user';
  
  return (
    <div className={`flex ${isUser ? 'justify-end' : 'justify-start'}`}>
      <div
        className={`max-w-[80%] p-3 ${isUser ? 'bg-accent-primary text-white' : 'bg-bg-secondary'}`}
        style={{ borderRadius: 0, boxShadow: '2px 2px 0 0 var(--shadow-color)' }}
      >
        {message.parts.map((part, i) => (
          <MessagePartRenderer key={i} part={part} />
        ))}
      </div>
    </div>
  );
}

function MessagePartRenderer({ part }: { part: MessagePart }) {
  switch (part.type) {
    case 'text':
      return <p className="whitespace-pre-wrap">{part.content}</p>;
    case 'code':
      return (
        <pre className="bg-bg-tertiary p-2 overflow-x-auto text-sm font-mono">
          <code>{part.content}</code>
        </pre>
      );
    default:
      return <span className="text-text-muted">[{part.type}]</span>;
  }
}
```
