Create a React hook that provides file AND project CRUD operations:

```typescript
import { useCallback } from 'react';
import { fileService } from '@/domain/services/file-service';
import { ProjectRegistry } from '@/domain/services/ProjectRegistry';
import { domainEventBus } from '@/infrastructure/events/domain-event-bus';
import { useFileTreeStore } from '@/infrastructure/persistence/stores/file-tree-store';
import { useShallow } from 'zustand/react/shallow';

export function useFileTreeOperations(projectId: string) {
  // Use useShallow per AGENTS.md mandate
  const { selectedPath, setSelectedPath } = useFileTreeStore(
    useShallow((s) => ({ selectedPath: s.selectedPath, setSelectedPath: s.selectFile }))
  );
  
  // ===== FILE OPERATIONS =====
  const createFile = useCallback(async (path: string, content = '') => {
    const result = await fileService.create(projectId, path, content);
    if (result.success) {
      setSelectedPath(path);
    }
    return result;
  }, [projectId, setSelectedPath]);
  
  const renameFile = useCallback(async (oldPath: string, newPath: string) => {
    // Read old content, create new file, delete old file
    const readResult = await fileService.read(projectId, oldPath);
    if (!readResult.success) return readResult;
    
    const createResult = await fileService.create(projectId, newPath, readResult.data);
    if (!createResult.success) return createResult;
    
    return fileService.delete(projectId, oldPath);
  }, [projectId]);
  
  const deleteFile = useCallback(async (path: string) => {
    return fileService.delete(projectId, path);
  }, [projectId]);
  
  // ===== PROJECT OPERATIONS (PLAT-01) =====
  const listProjects = useCallback(async () => {
    return ProjectRegistry.list();
  }, []);
  
  const createProject = useCallback(async (name: string) => {
    const result = await ProjectRegistry.create({ name, rootPath: `/${name}` });
    if (result.success) {
      domainEventBus.emit({ type: 'project:created', payload: result.data }, 'FileTree');
    }
    return result;
  }, []);
  
  const updateProject = useCallback(async (projectId: string, updates: { name?: string }) => {
    const result = await ProjectRegistry.update(projectId, updates);
    if (result.success) {
      domainEventBus.emit({ type: 'project:updated', payload: { projectId, ...updates } }, 'FileTree');
    }
    return result;
  }, []);
  
  const deleteProject = useCallback(async (targetProjectId: string) => {
    // First delete all files in the project
    const files = await fileService.listByProject(targetProjectId);
    for (const file of files) {
      await fileService.delete(targetProjectId, file.path);
    }
    // Then delete the project itself
    const result = await ProjectRegistry.delete(targetProjectId);
    if (result.success) {
      domainEventBus.emit({ type: 'project:deleted', payload: { projectId: targetProjectId } }, 'FileTree');
    }
    return result;
  }, []);
  
  // ===== PROJECT SWITCHING (PLAT-05) =====
  const switchProject = useCallback(async (targetProjectId: string) => {
    domainEventBus.emit({ type: 'project:switched', payload: { projectId: targetProjectId } }, 'FileTree');
  }, []);
  
  return {
    selectedPath,
    // File operations
    createFile,
    renameFile,
    deleteFile,
    // Project operations (PLAT-01)
    listProjects,
    createProject,
    updateProject,
    deleteProject,
    // Project switching (PLAT-05)
    switchProject,
  };
}
```

Key points:
- All file operations go through FileService (service-gated writes)
- All project operations go through ProjectRegistry
- Full CRUD for projects: list, create, update, delete (PLAT-01)
- switchProject fires project:switched event (PLAT-05)
- useShallow wrapper mandatory per AGENTS.md
- Returns CrudResult for consistent error handling
- projectId scopes all operations (NO workspaceId)

Create a project selector dropdown for switching between projects:

```typescript
import { useEffect, useState } from 'react';
import { useFileTreeOperations } from '../hooks/useFileTreeOperations';
import { useProjectStore } from '@/infrastructure/persistence/stores/project-store';
import { useShallow } from 'zustand/react/shallow';
import type { Project } from '@/domain/schemas/project.schema';

export function ProjectSelector() {
  const [projects, setProjects] = useState<Project[]>([]);
  const [isOpen, setIsOpen] = useState(false);
  
  const { activeProjectId, setActiveProject } = useProjectStore(
    useShallow((s) => ({ 
      activeProjectId: s.activeProjectId, 
      setActiveProject: s.setActiveProject 
    }))
  );
  
  const { listProjects, switchProject, createProject } = useFileTreeOperations(activeProjectId ?? '');
  
  useEffect(() => {
    listProjects().then(result => {
      if (result.success) {
        setProjects(result.data);
      }
    });
  }, [listProjects]);
  
  const activeProject = projects.find(p => p.id === activeProjectId);
  
  const handleSelect = async (projectId: string) => {
    setActiveProject(projectId);
    await switchProject(projectId);
    setIsOpen(false);
  };
  
  const handleCreateNew = async () => {
    const name = prompt('Project name:');
    if (name) {
      const result = await createProject(name);
      if (result.success) {
        await handleSelect(result.data.id);
        // Refresh list
        const listResult = await listProjects();
        if (listResult.success) {
          setProjects(listResult.data);
        }
      }
    }
  };
  
  return (
    <div className="project-selector">
      <button 
        onClick={() => setIsOpen(!isOpen)}
        className="project-selector-trigger"
      >
        <span>{activeProject?.name ?? 'Select Project'}</span>
        <ChevronDown className="chevron" />
      </button>
      
      {isOpen && (
        <div className="project-selector-dropdown">
          {projects.map(project => (
            <button
              key={project.id}
              onClick={() => handleSelect(project.id)}
              className={`project-item ${project.id === activeProjectId ? 'active' : ''}`}
            >
              {project.name}
            </button>
          ))}
          <div className="divider" />
          <button onClick={handleCreateNew} className="new-project">
            + New Project
          </button>
        </div>
      )}
    </div>
  );
}
```

- Solid colors, no gradients or blur
- Clear hover states

This component:
