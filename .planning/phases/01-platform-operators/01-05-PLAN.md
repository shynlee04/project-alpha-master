1. Ensure panel sizes always sum to 100%:
```typescript
setPanelSize(panelId: string, newSize: number): void {
  // Get all visible panels
  const visiblePanels = this.getVisiblePanels();
  const currentSize = this.panelSizes[panelId] ?? 0;
  const delta = newSize - currentSize;
  
  // Redistribute delta to other panels proportionally
  const otherPanels = visiblePanels.filter(p => p.id !== panelId);
  const otherTotal = otherPanels.reduce((sum, p) => sum + (this.panelSizes[p.id] ?? 0), 0);
  
  otherPanels.forEach(p => {
    const proportion = (this.panelSizes[p.id] ?? 0) / otherTotal;
    this.panelSizes[p.id] = Math.max(MIN_PANEL_SIZE, (this.panelSizes[p.id] ?? 0) - delta * proportion);
  });
  
  this.panelSizes[panelId] = Math.max(MIN_PANEL_SIZE, newSize);
  
  // Normalize to exactly 100%
  this.normalizePanelSizes();
}

private normalizePanelSizes(): void {
  const total = Object.values(this.panelSizes).reduce((sum, size) => sum + size, 0);
  if (total !== 100) {
    const factor = 100 / total;
    Object.keys(this.panelSizes).forEach(k => {
      this.panelSizes[k] *= factor;
    });
  }
}
```

```typescript
// src/presentation/components/layout/PanelResizer.tsx
import { useRef, useCallback, useEffect } from 'react';
import { usePluginLayoutStore } from '@/presentation/layouts/PluginLayoutStore';
import { useShallow } from 'zustand/react/shallow';

interface PanelResizerProps {
  panelId: string;
  direction: 'horizontal' | 'vertical';
  className?: string;
}

export function PanelResizer({ panelId, direction, className }: PanelResizerProps) {
  const { setPanelSize, panelSizes } = usePluginLayoutStore(
    useShallow(s => ({ setPanelSize: s.setPanelSize, panelSizes: s.panelSizes }))
  );
  
  const isDragging = useRef(false);
  const startPos = useRef(0);
  const startSize = useRef(0);
  const containerRef = useRef<HTMLDivElement | null>(null);
  
  const handleMouseDown = useCallback((e: React.MouseEvent) => {
    isDragging.current = true;
    startPos.current = direction === 'horizontal' ? e.clientX : e.clientY;
    startSize.current = panelSizes[panelId] ?? 20;
    
    // Prevent text selection during drag
    document.body.style.userSelect = 'none';
    document.body.style.cursor = direction === 'horizontal' ? 'col-resize' : 'row-resize';
  }, [direction, panelId, panelSizes]);
  
  useEffect(() => {
    const handleMouseMove = (e: MouseEvent) => {
      if (!isDragging.current) return;
      
      const container = containerRef.current?.parentElement?.parentElement;
      if (!container) return;
      
      const containerSize = direction === 'horizontal' 
        ? container.clientWidth 
        : container.clientHeight;
      
      const currentPos = direction === 'horizontal' ? e.clientX : e.clientY;
      const delta = currentPos - startPos.current;
      const deltaPercent = (delta / containerSize) * 100;
      
      const newSize = Math.min(80, Math.max(10, startSize.current + deltaPercent));
      setPanelSize(panelId, newSize);
    };
    
    const handleMouseUp = () => {
      isDragging.current = false;
      document.body.style.userSelect = '';
      document.body.style.cursor = '';
    };
    
    document.addEventListener('mousemove', handleMouseMove);
    document.addEventListener('mouseup', handleMouseUp);
    
    return () => {
      document.removeEventListener('mousemove', handleMouseMove);
      document.removeEventListener('mouseup', handleMouseUp);
    };
  }, [direction, panelId, setPanelSize]);
  
  return (
    <div
      ref={containerRef}
      onMouseDown={handleMouseDown}
      className={`${direction === 'horizontal' ? 'w-1 cursor-col-resize' : 'h-1 cursor-row-resize'} bg-border-primary hover:bg-accent-primary transition-colors ${className ?? ''}`}
      style={{ borderRadius: 0 }} // 8-bit: sharp corners
    />
  );
}
```

Key points:
- Handles both horizontal and vertical resizing
- Uses percentage-based sizing relative to container
- Clamps to min/max (10% - 80%)
- Cleans up event listeners on unmount
- useShallow for Zustand per AGENTS.md

```typescript
togglePlugin(pluginId: string): void {
  set(state => {
    const isActive = state.activePlugins.has(pluginId);
    const newActive = new Set(state.activePlugins);
    
    if (isActive) {
      newActive.delete(pluginId);
      // Update layout to hide panel
      usePluginLayoutStore.getState().setPanelVisible(pluginId, false);
    } else {
      newActive.add(pluginId);
      // Update layout to show panel
      usePluginLayoutStore.getState().setPanelVisible(pluginId, true);
    }
    
    return { activePlugins: newActive };
  });
}
```

```typescript
setPanelVisible(panelId: string, visible: boolean): void {
  if (visible) {
    // Show panel with default size if hidden
    if (!this.visiblePanels.has(panelId)) {
      this.visiblePanels.add(panelId);
      this.panelSizes[panelId] = 20; // Default 20%
      this.normalizePanelSizes();
    }
  } else {
    // Hide panel and redistribute space
    if (this.visiblePanels.has(panelId)) {
      this.visiblePanels.delete(panelId);
      delete this.panelSizes[panelId];
      this.normalizePanelSizes();
    }
  }
}
```
