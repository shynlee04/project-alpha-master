import type { StorageAdapter, FileMetadata } from './storage-types';
import { db } from '@/infrastructure/persistence/dexie-db';

export class IDBStorageAdapter implements StorageAdapter {
  readonly name = 'idb';
  
  static isSupported(): boolean {
    return typeof indexedDB !== 'undefined';
  }
  
  async write(projectId: string, path: string, content: string): Promise<FileMetadata> {
    const now = new Date().toISOString();
    const record = {
      id: `${projectId}:${path}`,
      projectId,
      path,
      content,
      createdAt: now,
      updatedAt: now,
    };
    
    await db.files.put(record);
    
    return {
      path,
      size: content.length,
      updatedAt: now,
    };
  }
  
  async read(projectId: string, path: string): Promise<string | null> {
    const record = await db.files.get(`${projectId}:${path}`);
    return record?.content ?? null;
  }
  
  async delete(projectId: string, path: string): Promise<void> {
    await db.files.delete(`${projectId}:${path}`);
  }
  
  async list(projectId: string, dirPath: string): Promise<FileMetadata[]> {
    const prefix = `${projectId}:${dirPath}`;
    const records = await db.files
      .where('id')
      .startsWith(prefix)
      .toArray();
    
    return records.map(r => ({
      path: r.path,
      size: r.content.length,
      updatedAt: r.updatedAt,
    }));
  }
  
  async exists(projectId: string, path: string): Promise<boolean> {
    const record = await db.files.get(`${projectId}:${path}`);
    return record !== undefined;
  }
}
```

Key points:
- Uses Dexie db.files table for storage
- Composite key: `${projectId}:${path}` ensures project isolation
- No workspaceId - only projectId per NO-WORKSPACE-MANDATE
- Returns consistent FileMetadata format

  // Check if we have a stored handle in Dexie
  const storedHandle = await db.fsaHandles.get(projectId);
  if (!storedHandle) return false;
  
  // Verify permission is still granted
  const permissionStatus = await storedHandle.handle.queryPermission({ mode: 'readwrite' });
  if (permissionStatus === 'granted') {
    this.directoryHandle = storedHandle.handle;
    return true;
  }
  
  // Permission lost - need to re-request
  return false;
}

// src/infrastructure/filesystem/StorageAdapterFactory.ts
import { FSAStorageAdapter } from './fsa-storage-adapter';
import { IDBStorageAdapter } from './idb-storage-adapter';
import type { StorageAdapter } from './storage-types';

export class StorageAdapterFactory {
  private static cache = new Map<string, StorageAdapter>();
  
  static getAdapter(projectId: string): StorageAdapter {
    if (this.cache.has(projectId)) {
      return this.cache.get(projectId)!;
    }
    
    // Platform detection: FSA on desktop, IDB on mobile
    const adapter = FSAStorageAdapter.isSupported()
      ? new FSAStorageAdapter(projectId)
      : new IDBStorageAdapter(projectId);
    
    this.cache.set(projectId, adapter);
    return adapter;
  }
  
  static async restoreOrRequest(projectId: string): Promise<StorageAdapter> {
    const adapter = this.getAdapter(projectId);
    
    if (adapter instanceof FSAStorageAdapter) {
      const restored = await adapter.restoreHandle(projectId);
      if (!restored) {
        // Will need permission request in UI
        console.warn('FSA handle needs permission restore for project:', projectId);
      }
    }
    
    return adapter;
  }
  
  static clearCache(): void {
    this.cache.clear();
  }
}

export const getStorageAdapter = StorageAdapterFactory.getAdapter.bind(StorageAdapterFactory);
```
