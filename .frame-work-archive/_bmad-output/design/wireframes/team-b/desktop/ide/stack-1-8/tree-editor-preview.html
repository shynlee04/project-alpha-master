import React, { useState } from 'react'; import { useTodo } from './useTodo'; import { TodoItem } from './TodoItem'; interface TodoListProps { title?: string; initialItems?: string[]; } export const TodoList: React.FC<TodoListProps> = ({ title = 'My Tasks', initialItems = [], }) => { const { todos, addTodo, toggleTodo, deleteTodo, } = useTodo(initialItems); const [newTodo, setNewTodo] = useState(''); const handleSubmit = (e: React.FormEvent) => { e.preventDefault(); if (newTodo.trim()) { addTodo(newTodo.trim()); setNewTodo(''); } }; return ( <div className="todo-list"> <h2>{title}</h2> <form onSubmit={handleSubmit}> <input type="text" value={newTodo} onChange={(e) => setNewTodo(e.target.value)} placeholder="Add a new task..." /> <button type="submit">Add</button> </form> <ul> {todos.map((todo) => ( <TodoItem key={todo.id} todo={todo} onToggle={() => toggleTodo(todo.id)} onDelete={() => deleteTodo(todo.id)} /> ))} </ul> </div> ); };
