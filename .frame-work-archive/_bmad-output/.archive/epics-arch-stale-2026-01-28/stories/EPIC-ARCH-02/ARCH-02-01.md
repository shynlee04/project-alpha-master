## Implementation Notes

### File 1: src/domain/interfaces/feature-plugin.interface.ts

```typescript
// Core interface
export interface FeaturePlugin {
  // Identity
  id: PluginId;
  name: string;
  icon: React.ReactNode;
  description: string;

  // Requirements
  requirements: {
    storageType: 'fsa' | 'indexeddb' | 'any';
    deviceType: 'desktop' | 'mobile' | 'any';
    minWidth: number;
    maxInstances: 1 | 2 | 'unlimited';
  };

  // Rendering
  MainComponent: React.FC<PluginMainProps>;
  SidebarComponent?: React.FC<PluginSidebarProps>;
  ToolbarComponent?: React.FC<PluginToolbarProps>;

  // Lifecycle
  onMount?: (context: ProjectContext) => Promise<void>;
  onUnmount?: () => Promise<void>;
  onProjectChange?: (newProjectId: string) => Promise<void>;
}

// Props interfaces
export interface PluginMainProps {
  projectContext: ProjectContext;
  panelId: string;
  width: number;
  height: number;
}

export interface PluginSidebarProps {
  projectContext: ProjectContext;
  collapsed: boolean;
  onToggle: () => void;
}

export interface PluginToolbarProps {
  projectContext: ProjectContext;
  onAction: (action: string) => void;
}

// Note: ProjectContext is a forward reference - will be defined in ARCH-02-03
// For now, declare it as an interface without definition
export interface ProjectContext {
  // To be filled in ARCH-02-03
  [key: string]: any;
}
```
