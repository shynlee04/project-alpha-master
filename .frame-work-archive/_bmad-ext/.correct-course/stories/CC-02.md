## Implementation Notes

### Install DOMPurify
```bash
pnpm add dompurify

```

### Sanitization Utilities
```typescript
// src/infrastructure/security/sanitization.ts

import DOMPurify from 'dompurify';

// Default sanitization for user-provided content
export function sanitizeHtml(html: string): string {
  return DOMPurify.sanitize(html, {
    ALLOWED_TAGS: ['h1', 'h2', 'h3', 'p', 'strong', 'em', 'code', 'pre', 'ul', 'ol', 'li'],
    ALLOWED_ATTR: ['href', 'title', 'class'],
    ALLOW_DATA_ATTR: false,
  });
}

// For Mermaid SVG (trusted source but external)
export function sanitizeSvg(svg: string): string {
  return DOMPurify.sanitize(svg, {
    ALLOWED_TAGS: ['svg', 'g', 'path', 'rect', 'circle', 'text'],
    ALLOWED_ATTR: ['class', 'd', 'fill', 'stroke'],
  });
}

// For iframe srcdoc (most restrictive)
export function sanitizeIframeContent(html: string): string {
  return DOMPurify.sanitize(html, {
    ALLOWED_TAGS: ['div', 'span', 'p', 'br'],
    ALLOWED_ATTR: ['class', 'style'],
    ALLOW_DATA_ATTR: false,
  });
}
```
